Identificação do projeto

Escopo
Banco de dados relacional da Pichau Informática, varejista de informática com venda online. Cobre clientes e endereços de entrega, usuários e perfis, categorias, catálogo com histórico de preços, controle de estoque, formas de pagamento, pedidos com cancelamento e estorno, consentimento e anonimização LGPD e trilha de auditoria. Ficam fora: marketplaces, processamento de pagamento, fornecedores, folha e emissão de NF-e.
Tecnologia
MySQL 8.0, engine InnoDB, utf8mb4. Orientador: Prof. Dr. Diego Carvalho.
Responsáveis
  • Erwin Arantes da Silva Klemm - Líder de Requisitos, Data Steward. Qualidade técnica, dicionário de dados, padronização e rastreabilidade (RF, RNF, RN, PR).
  • Pedro Neri - Analista de Negócio e Dados, Data Owner. Acurácia das regras de negócio (RN), das políticas (PR) e da estrutura lógica das entidades.

-- Convenções: snake_case, tabelas no singular, PK id_<entidade>, FK com o nome da PK referenciada, exclusão lógica por ativo.

SET NAMES utf8mb4;

-- Agrupa os produtos do catálogo (RF04).
CREATE TABLE categoria (
  id_categoria INT NOT NULL AUTO_INCREMENT,
  nome VARCHAR(100) NOT NULL,
  ativo TINYINT(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (id_categoria),
  UNIQUE KEY uk_categoria_nome (nome)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Agrupa os produtos do catálogo (RF04).';

-- Domínio das formas de pagamento aceitas, normalizado fora do pedido (3FN).
CREATE TABLE forma_pagamento (
  id_forma_pagamento INT NOT NULL AUTO_INCREMENT,
  descricao VARCHAR(50) NOT NULL,
  ativo TINYINT(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (id_forma_pagamento),
  UNIQUE KEY uk_forma_pagamento_descricao (descricao)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Domínio das formas de pagamento aceitas, normalizado fora do pedido (3FN).';

-- Pessoa física ou jurídica que compra produtos (RF01, RF12).
CREATE TABLE cliente (
  id_cliente INT NOT NULL AUTO_INCREMENT,
  nome VARCHAR(150) NOT NULL,
  cpf_cnpj VARCHAR(14) NOT NULL,
  telefone VARCHAR(15) NOT NULL,
  email VARCHAR(150) NULL,
  data_consentimento DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  data_anonimizacao DATETIME NULL,
  ativo TINYINT(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (id_cliente),
  UNIQUE KEY uk_cliente_cpf_cnpj (cpf_cnpj)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Pessoa física ou jurídica que compra produtos (RF01, RF12).';

-- Operador do sistema com perfil de acesso (RF03).
CREATE TABLE usuario (
  id_usuario INT NOT NULL AUTO_INCREMENT,
  nome VARCHAR(150) NOT NULL,
  login VARCHAR(50) NOT NULL,
  senha_hash VARCHAR(255) NOT NULL,
  perfil ENUM('ATENDENTE','GERENTE','ADMINISTRADOR') NOT NULL,
  ativo TINYINT(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (id_usuario),
  UNIQUE KEY uk_usuario_login (login)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Operador do sistema com perfil de acesso (RF03).';

-- Endereço de entrega de um cliente, com campos atômicos (1FN).
CREATE TABLE endereco (
  id_endereco INT NOT NULL AUTO_INCREMENT,
  id_cliente INT NOT NULL,
  logradouro VARCHAR(150) NOT NULL,
  numero VARCHAR(10) NOT NULL,
  complemento VARCHAR(60) NULL,
  bairro VARCHAR(80) NOT NULL,
  cep CHAR(8) NOT NULL,
  cidade VARCHAR(80) NOT NULL,
  uf CHAR(2) NOT NULL,
  ativo TINYINT(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (id_endereco),
  CONSTRAINT chk_endereco_1 CHECK (CHAR_LENGTH(cep) = 8),
  CONSTRAINT chk_endereco_2 CHECK (CHAR_LENGTH(uf) = 2),
  CONSTRAINT fk_endereco_id_cliente FOREIGN KEY (id_cliente) REFERENCES cliente (id_cliente) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Endereço de entrega de um cliente, com campos atômicos (1FN).';

-- Item comercializado, com preço de tabela e saldo físico (RF05, RF06).
CREATE TABLE produto (
  id_produto INT NOT NULL AUTO_INCREMENT,
  id_categoria INT NOT NULL,
  descricao VARCHAR(200) NOT NULL,
  preco_tabela DECIMAL(10,2) NOT NULL,
  saldo_estoque INT NOT NULL DEFAULT 0,
  ativo TINYINT(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (id_produto),
  CONSTRAINT chk_produto_1 CHECK (preco_tabela > 0),
  CONSTRAINT chk_produto_2 CHECK (saldo_estoque >= 0),
  CONSTRAINT fk_produto_id_categoria FOREIGN KEY (id_categoria) REFERENCES categoria (id_categoria) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Item comercializado, com preço de tabela e saldo físico (RF05, RF06).';

-- Registra cada alteração de preço de tabela (RF05, RN10).
CREATE TABLE historico_preco (
  id_historico_preco INT NOT NULL AUTO_INCREMENT,
  id_produto INT NOT NULL,
  id_usuario INT NOT NULL,
  preco_anterior DECIMAL(10,2) NOT NULL,
  preco_novo DECIMAL(10,2) NOT NULL,
  data_alteracao DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id_historico_preco),
  CONSTRAINT chk_historico_preco_1 CHECK (preco_anterior > 0),
  CONSTRAINT chk_historico_preco_2 CHECK (preco_novo > 0),
  CONSTRAINT fk_historico_preco_id_produto FOREIGN KEY (id_produto) REFERENCES produto (id_produto) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_historico_preco_id_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Registra cada alteração de preço de tabela (RF05, RN10).';

-- Venda registrada para um cliente, com ciclo ABERTO, FINALIZADO e CANCELADO (RF07, RF08).
CREATE TABLE pedido (
  id_pedido INT NOT NULL AUTO_INCREMENT,
  id_cliente INT NOT NULL,
  id_usuario INT NOT NULL,
  id_forma_pagamento INT NOT NULL,
  id_endereco INT NOT NULL,
  data_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  situacao ENUM('ABERTO','FINALIZADO','CANCELADO') NOT NULL DEFAULT 'ABERTO',
  valor_desconto DECIMAL(10,2) NOT NULL DEFAULT 0.00,
  valor_total DECIMAL(10,2) NOT NULL DEFAULT 0.00,
  data_cancelamento DATETIME NULL,
  PRIMARY KEY (id_pedido),
  CONSTRAINT chk_pedido_1 CHECK (valor_desconto >= 0),
  CONSTRAINT chk_pedido_2 CHECK (valor_total >= 0),
  CONSTRAINT fk_pedido_id_cliente FOREIGN KEY (id_cliente) REFERENCES cliente (id_cliente) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_pedido_id_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_pedido_id_forma_pagamento FOREIGN KEY (id_forma_pagamento) REFERENCES forma_pagamento (id_forma_pagamento) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_pedido_id_endereco FOREIGN KEY (id_endereco) REFERENCES endereco (id_endereco) ON DELETE RESTRICT ON UPDATE CASCADE,
  KEY idx_pedido_data_hora (data_hora),
  KEY idx_pedido_id_cliente (id_cliente),
  KEY idx_pedido_id_usuario (id_usuario)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Venda registrada para um cliente, com ciclo ABERTO, FINALIZADO e CANCELADO (RF07, RF08).';

-- Entidade associativa que resolve o N:N entre pedido e produto (RF07).
CREATE TABLE item_pedido (
  id_item_pedido INT NOT NULL AUTO_INCREMENT,
  id_pedido INT NOT NULL,
  id_produto INT NOT NULL,
  quantidade INT NOT NULL,
  preco_unitario_venda DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id_item_pedido),
  CONSTRAINT chk_item_pedido_1 CHECK (quantidade > 0),
  CONSTRAINT chk_item_pedido_2 CHECK (preco_unitario_venda >= 0),
  CONSTRAINT fk_item_pedido_id_pedido FOREIGN KEY (id_pedido) REFERENCES pedido (id_pedido) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_item_pedido_id_produto FOREIGN KEY (id_produto) REFERENCES produto (id_produto) ON DELETE RESTRICT ON UPDATE CASCADE,
  UNIQUE KEY uk_item_pedido_pedido_produto (id_pedido, id_produto)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Entidade associativa que resolve o N:N entre pedido e produto (RF07).';

-- Estorno integral de pedido cancelado dentro do prazo legal (RF08, PR02).
CREATE TABLE estorno (
  id_estorno INT NOT NULL AUTO_INCREMENT,
  id_pedido INT NOT NULL,
  id_usuario INT NOT NULL,
  valor_estornado DECIMAL(10,2) NOT NULL,
  data_estorno DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id_estorno),
  UNIQUE KEY uk_estorno_id_pedido (id_pedido),
  CONSTRAINT chk_estorno_1 CHECK (valor_estornado > 0),
  CONSTRAINT fk_estorno_id_pedido FOREIGN KEY (id_pedido) REFERENCES pedido (id_pedido) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_estorno_id_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='Estorno integral de pedido cancelado dentro do prazo legal (RF08, PR02).';

-- Trilha de auditoria append-only das operações de alteração e exclusão (RF11, RNF01).
CREATE TABLE log_auditoria (
